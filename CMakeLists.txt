# The core is split into independent libraries (specification §2.1, §6, §9.2):
#   ckey_codec      encode and decode codewords (RS, CRC, framing)
#   ckey_scan       image localization and decoding; needs only ckey_codec
#   ckey_render     geometry, SVG, raster, PNG, PDF, print; needs nothing else
#   ckey_certify    optical certification through a modelled scan; needs scan + render
#   ckey_telemetry  optional aggregate counters; nothing depends on it
# A scan-only app links ckey_scan alone. ckey_core is codec, scan, render and certify. The
# Xcode projects compile src/ckey.cpp, the amalgamation of the same parts.
function(ckey_library name source)
  add_library(${name} ${source})
  target_compile_features(${name} PUBLIC cxx_std_20)
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  set_target_properties(${name} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE YES)
  if(BUILD_SHARED_LIBS)
    target_compile_definitions(${name} PRIVATE CKEY_SHARED CKEY_BUILDING)
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    # Each library compiles only the shared internals it calls.
    target_compile_options(${name} PRIVATE -Wno-unused-function)
  endif()
endfunction()
ckey_library(ckey_codec src/ckey_codec.cpp)
ckey_library(ckey_scan src/ckey_scan.cpp)
target_link_libraries(ckey_scan PUBLIC ckey_codec)
ckey_library(ckey_render src/ckey_render.cpp)
ckey_library(ckey_certify src/ckey_certify.cpp)
target_link_libraries(ckey_certify PUBLIC ckey_scan ckey_render)
add_library(ckey_core INTERFACE)
target_link_libraries(ckey_core INTERFACE ckey_codec ckey_scan ckey_render ckey_certify)

# ---- Install and CMake package (spec §7: C/C++ consumers use find_package(CKey)) ----
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/VERSION CKEY_VERSION_STRING LIMIT_COUNT 1)
set(CKEY_INSTALL_TARGETS ckey_codec ckey_scan ckey_render ckey_certify ckey_core)
if(CKEY_BUILD_TELEMETRY)
  list(APPEND CKEY_INSTALL_TARGETS ckey_telemetry)
endif()
foreach(lib ${CKEY_INSTALL_TARGETS})
  string(REPLACE "ckey_" "" short ${lib})
  set_target_properties(${lib} PROPERTIES EXPORT_NAME ${short})
endforeach()
install(TARGETS ${CKEY_INSTALL_TARGETS} EXPORT CKeyTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} INCLUDES DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS ckey RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ckey DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CKeyTargets NAMESPACE CKey:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CKey)
configure_package_config_file(cmake/CKeyConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/CKeyConfig.cmake
                              INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CKey)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/CKeyConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/CKeyConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/CKeyConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CKey)
