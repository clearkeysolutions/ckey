cmake_minimum_required(VERSION 3.20)
project(CKey VERSION 0.1.0 LANGUAGES CXX)

option(CKEY_BUILD_TESTS "Build CKey tests" ON)
add_library(ckey_core src/ckey.cpp)
target_compile_features(ckey_core PUBLIC cxx_std_20)
target_include_directories(ckey_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
set_target_properties(ckey_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE YES)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(ckey_core PRIVATE CKEY_SHARED CKEY_BUILDING)
endif()
add_executable(ckey cli/main.cpp)
target_link_libraries(ckey PRIVATE ckey_core)
if(CKEY_BUILD_TESTS)
  enable_testing()
  add_executable(ckey_tests tests/core_tests.cpp)
  target_link_libraries(ckey_tests PRIVATE ckey_core)
  add_test(NAME ckey_core_tests COMMAND ckey_tests)
  add_executable(ckey_image_tests tests/image_decode_tests.cpp)
  target_link_libraries(ckey_image_tests PRIVATE ckey_core)
  target_compile_definitions(ckey_image_tests PRIVATE CKEY_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME ckey_optical_image_decode COMMAND ckey_image_tests)
  add_executable(ckey_roundtrip_tests tests/roundtrip_tests.cpp)
  target_link_libraries(ckey_roundtrip_tests PRIVATE ckey_core)
  add_test(NAME ckey_render_decode_roundtrip COMMAND ckey_roundtrip_tests)
  add_executable(ckey_reader_pipeline_tests tests/reader_pipeline_tests.cpp)
  target_link_libraries(ckey_reader_pipeline_tests PRIVATE ckey_core)
  add_test(NAME ckey_simulated_reader COMMAND ckey_reader_pipeline_tests)
  # White-box: Profile 0 layout invariants, capacities, ECC, and every version's image round trip.
  add_executable(ckey_scalable_tests tests/scalable_tests.cpp)
  target_include_directories(ckey_scalable_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(ckey_scalable_tests PRIVATE cxx_std_20)
  add_test(NAME ckey_scalable_profile0 COMMAND ckey_scalable_tests)
  # White-box: compiles the core translation unit directly to inspect layout.
  add_executable(ckey_layout_tests tests/layout_tests.cpp)
  target_include_directories(ckey_layout_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(ckey_layout_tests PRIVATE cxx_std_20)
  add_test(NAME ckey_dense_layout COMMAND ckey_layout_tests)
  foreach(test_target ckey_tests ckey_image_tests ckey_roundtrip_tests ckey_layout_tests ckey_reader_pipeline_tests ckey_scalable_tests)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${test_target} PRIVATE -Wall -Wextra)
    endif()
  endforeach()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME ckey_independent_rs_vector COMMAND ${Python3_EXECUTABLE} tools/rs_reference.py)
    set_tests_properties(ckey_independent_rs_vector PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME ckey_version_consistency COMMAND ${Python3_EXECUTABLE} tools/check_version.py)
    set_tests_properties(ckey_version_consistency PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
